cmake_minimum_required(VERSION 3.16)
project(anduinos_recovery_gui C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_SKIP_RPATH TRUE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_FETCH_DEPENDENCIES OFF CACHE BOOL "" FORCE)
set(LV_USE_FIND_PACKAGE OFF CACHE BOOL "" FORCE)
set(LV_USE_PKG_CONFIG OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH ${CMAKE_SOURCE_DIR}/src/lv_conf.h CACHE PATH "" FORCE)
add_subdirectory(${LVGL_SOURCE} lvgl)
set(INPUT_LIB ${INPUT_ROOT}/usr/lib/${TARGET_TRIPLET})
target_include_directories(lvgl PRIVATE ${INPUT_ROOT}/usr/include/libdrm ${INPUT_ROOT}/usr/include/freetype2 ${INPUT_ROOT}/usr/include)
target_compile_options(lvgl PRIVATE -Os -ffunction-sections -fdata-sections)
add_executable(recovery-gui src/gui.c)
target_include_directories(recovery-gui PRIVATE ${CMAKE_BINARY_DIR} ${INPUT_ROOT}/usr/include)
target_compile_options(recovery-gui PRIVATE -Wall -Wextra -Werror)
target_link_libraries(recovery-gui PRIVATE lvgl ${INPUT_LIB}/libdrm.so.2 ${INPUT_LIB}/libfreetype.so.6 ${INPUT_LIB}/libjq.so.1 m)
target_link_options(recovery-gui PRIVATE -Wl,--gc-sections -Wl,-rpath-link,${INPUT_LIB})
